WITHOUT IN : 

SELECT * FROM student where city = 'pune' OR city ='goa' OR city = 'nsk' OR city = 'thane';

With IN :

SELECT * FROM student where city IN('pune','goa','nsk','thane');

SELECT * FROM student where city NOT IN('pune','goa','nsk','thane');


WITHOUT BETWEEN : range : 

SELECT * FROM employee where salary >= 30000 AND salary <= 70000;

WITH BETWEEN  : 

SELECT * FROM employee where salary BETWEEN 30000 AND 70000;

SELECT * FROM employee where salary NOT BETWEEN 30000 AND 70000;

SELECT * FROM employee where salary BETWEEN 30000 AND 50000 AND salary != 40000;
